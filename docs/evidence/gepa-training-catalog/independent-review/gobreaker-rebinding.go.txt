package gobreaker

import "testing"

func TestIndependentCallerRebinding(t *testing.T) {
    a := NewTwoStepCircuitBreaker(Settings{Name: "a"})
    b := NewTwoStepCircuitBreaker(Settings{Name: "b"})
    original := a
    done, err := a.Allow()
    if err != nil { t.Fatal(err) }
    a = b
    if a != b || original == a { t.Fatal("pointer rebinding precondition") }
    done(true)
    if got := original.Counts(); got.Requests != 1 || got.TotalSuccesses != 1 { t.Fatalf("original: %+v", got) }
    if got := b.Counts(); got.Requests != 0 || got.TotalSuccesses != 0 { t.Fatalf("replacement: %+v", got) }
}
