package gobreaker
import "testing"

// Counterfactual probe of routed Q-08's unqualified original-breaker claim.
// Expected red: this is not an approved behavior requirement.
func TestArchivedOriginalBindingClaim(t *testing.T) {
 a := NewTwoStepCircuitBreaker(Settings{Name:"a"})
 b := NewTwoStepCircuitBreaker(Settings{Name:"b"})
 done,err:=a.Allow(); if err!=nil {t.Fatal(err)}
 *a=*b
 done(true)
 if got:=b.Counts(); got.Requests!=0 || got.TotalSuccesses!=0 {
  t.Fatalf("archived original-binding claim refuted: replacement counts=%+v",got)
 }
}
