func probeHalfOpen(t *testing.T, b *TwoStepCircuitBreaker) {
	t.Helper()
	d, e := b.Allow()
	if e != nil {
		t.Fatal(e)
	}
	d(false)
	// Scheduling only. Move the existing deadline into the past.
	b.cb.mutex.Lock()
	b.cb.expiry = time.Now().Add(-time.Second)
	b.cb.mutex.Unlock()
	if got := b.State(); got != StateHalfOpen {
		t.Fatalf("state = %v", got)
	}
}

func TestPilotDuplicateAndMissing(t *testing.T) {
	b := NewTwoStepCircuitBreaker(Settings{MaxRequests: 2, ReadyToTrip: func(Counts) bool { return true }})
	probeHalfOpen(t, b)
	a, err := b.Allow()
	if err != nil {
		t.Fatal(err)
	}
	later, err := b.Allow()
	if err != nil {
		t.Fatal(err)
	}
	if _, err := b.Allow(); err != ErrTooManyRequests {
		t.Fatalf("at capacity: %v", err)
	}
	a(true)
	if got := b.Counts(); got.Requests != 2 || got.TotalSuccesses != 1 {
		t.Fatalf("after one report: %+v", got)
	}
	if _, err := b.Allow(); err != ErrTooManyRequests {
		t.Fatalf("held slot: %v", err)
	}
	a(true)
	if got := b.State(); got != StateClosed {
		t.Fatalf("duplicate did not close: %v", got)
	}
	later(false)
	if got := b.Counts(); got != (Counts{}) {
		t.Fatalf("stale failure counted: %+v", got)
	}
}

func TestPilotStaleCompletionRefresh(t *testing.T) {
	notifications := make(chan State, 3)
	b := NewTwoStepCircuitBreaker(Settings{ReadyToTrip: func(Counts) bool { return true }, OnStateChange: func(_ string, _, to State) { notifications <- to }})
	old, _ := b.Allow()
	trip, _ := b.Allow()
	trip(false)
	if got := <-notifications; got != StateOpen {
		t.Fatal(got)
	}
	b.cb.mutex.Lock()
	b.cb.expiry = time.Now().Add(-time.Second)
	b.cb.mutex.Unlock()
	_ = b.Counts()
	select {
	case got := <-notifications:
		t.Fatalf("Counts notified %v", got)
	default:
	}
	old(true)
	select {
	case got := <-notifications:
		if got != StateHalfOpen {
			t.Fatal(got)
		}
	default:
		t.Fatal("stale report failed to refresh")
	}
	if got := b.Counts(); got != (Counts{}) {
		t.Fatalf("stale report counted: %+v", got)
	}
}

func TestPilotPolicyPanicAccounting(t *testing.T) {
	const marker = "policy panic"
	b := NewCircuitBreaker(Settings{ReadyToTrip: func(Counts) bool { panic(marker) }})
	func() {
		defer func() {
			if got := recover(); got != marker {
				t.Fatalf("panic = %v", got)
			}
		}()
		_, _ = b.Execute(func() (interface{}, error) { return nil, errors.New("request error") })
	}()
	if got := b.Counts(); got.Requests != 1 || got.TotalFailures != 2 {
		t.Fatalf("counts = %+v", got)
	}
}

func TestPilotNotifyPanicPrecedence(t *testing.T) {
	b := NewCircuitBreaker(Settings{ReadyToTrip: func(Counts) bool { return true }, OnStateChange: func(string, State, State) { panic("notification panic") }})
	func() {
		defer func() {
			if got := recover(); got != "notification panic" {
				t.Fatalf("panic = %v", got)
			}
		}()
		_, _ = b.Execute(func() (interface{}, error) { panic("request panic") })
	}()
	if b.State() != StateOpen || b.Counts() != (Counts{}) {
		t.Fatal("state transition did not commit before panic")
	}
}

func TestPilotNotifyReentry(t *testing.T) {
	entered := make(chan struct{})
	returned := make(chan struct{})
	var b *CircuitBreaker
	b = NewCircuitBreaker(Settings{ReadyToTrip: func(Counts) bool { return true }, OnStateChange: func(string, State, State) {
		close(entered)
		_ = b.State()
		close(returned)
	}})
	go func() { _, _ = b.Execute(func() (interface{}, error) { return nil, errors.New("trip") }) }()
	select {
	case <-entered:
	case <-time.After(time.Second):
		t.Fatal("hook did not start")
	}
	select {
	case <-returned:
		t.Fatal("lock-taking reentry returned")
	case <-time.After(40 * time.Millisecond):
	}
}

func TestPilotClassifierWindow(t *testing.T) {
	entered, release, complete := make(chan struct{}), make(chan struct{}), make(chan error, 1)
	b := NewCircuitBreaker(Settings{Interval: time.Hour, IsSuccessful: func(error) bool { close(entered); <-release; return true }})
	go func() { _, err := b.Execute(func() (interface{}, error) { return "original", nil }); complete <- err }()
	select {
	case <-entered:
	case <-time.After(time.Second):
		t.Fatal("classifier did not start")
	}
	b.mutex.Lock()
	b.expiry = time.Now().Add(-time.Second)
	b.mutex.Unlock()
	if before := b.Counts(); before.Requests != 1 {
		t.Fatalf("before reset: %+v", before)
	}
	if b.State() != StateClosed {
		t.Fatal("unexpected state")
	}
	close(release)
	select {
	case err := <-complete:
		if err != nil {
			t.Fatal(err)
		}
	case <-time.After(time.Second):
		t.Fatal("request did not finish")
	}
	if got := b.Counts(); got != (Counts{}) {
		t.Fatalf("late classifier counted: %+v", got)
	}
}

func TestPilotGoexit(t *testing.T) {
	b := NewCircuitBreaker(Settings{})
	exited := make(chan struct{})
	go func() {
		defer close(exited)
		_, _ = b.Execute(func() (interface{}, error) { runtime.Goexit(); return nil, nil })
	}()
	select {
	case <-exited:
	case <-time.After(time.Second):
		t.Fatal("goroutine did not exit")
	}
	if got := b.Counts(); got != (Counts{Requests: 1}) {
		t.Fatalf("exit counts: %+v", got)
	}
}

func TestPilotSecondExecuteCompletesFirst(t *testing.T) {
	b := NewCircuitBreaker(Settings{})
	firstStarted, release := make(chan struct{}), make(chan struct{})
	firstResult := make(chan interface{}, 1)
	go func() {
		value, _ := b.Execute(func() (interface{}, error) {
			close(firstStarted)
			<-release
			return "first", nil
		})
		firstResult <- value
	}()
	select {
	case <-firstStarted:
	case <-time.After(time.Second):
		t.Fatal("first did not start")
	}
	second, err := b.Execute(func() (interface{}, error) { return "second", nil })
	if err != nil || second != "second" {
		t.Fatalf("second = %v, %v", second, err)
	}
	if got := b.Counts(); got != (Counts{Requests: 2, TotalSuccesses: 1, ConsecutiveSuccesses: 1}) {
		t.Fatalf("first still pending: %+v", got)
	}
	close(release)
	select {
	case value := <-firstResult:
		if value != "first" {
			t.Fatal(value)
		}
	case <-time.After(time.Second):
		t.Fatal("first did not return")
	}
}

func TestPilotRecoveryNotificationPanicBeforeAdmission(t *testing.T) {
	b := NewCircuitBreaker(Settings{
		ReadyToTrip: func(Counts) bool { return true },
		OnStateChange: func(_ string, _, to State) {
			if to == StateHalfOpen {
				panic("half-open notification")
			}
		},
	})
	_, _ = b.Execute(func() (interface{}, error) { return nil, errors.New("trip") })
	b.mutex.Lock()
	b.expiry = time.Now().Add(-time.Second)
	b.mutex.Unlock()
	func() {
		defer func() {
			if got := recover(); got != "half-open notification" {
				t.Fatalf("panic = %v", got)
			}
		}()
		_, _ = b.Execute(func() (interface{}, error) { t.Fatal("request ran before notification returned"); return nil, nil })
	}()
	if b.State() != StateHalfOpen || b.Counts() != (Counts{}) {
		t.Fatal("transition or admission commitment differed")
	}
}
